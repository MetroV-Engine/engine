# The engine: where the modules of src/ are connected together.
#   - engine_ecs    (src/ECS):    the world, its components and systems
#   - engine_editor (src/Editor): the editor GUI, opened on that world
# Two ways to build it:
#  - from the engine root, with -DENGINE_BUILD_GUI=ON (see the root CMakeLists.txt);
#  - from this folder alone (cmake -S . -B build in src/Engine): it then adds
#    the ECS and the editor itself and uses the engine's vcpkg manifest with
#    the "gui" feature.
cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  # vcpkg toolchain, when not given with -DCMAKE_TOOLCHAIN_FILE: the engine's
  # vcpkg submodule if present, else the one in $VCPKG_ROOT.
  if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../vcpkg/scripts/buildsystems/vcpkg.cmake")
      set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../vcpkg/scripts/buildsystems/vcpkg.cmake"
          CACHE STRING "Vcpkg toolchain file")
    elseif(DEFINED ENV{VCPKG_ROOT})
      set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake"
          CACHE STRING "Vcpkg toolchain file")
    endif()
  endif()

  # vcpkg reads these at project(): the manifest is at the repository root.
  set(VCPKG_MANIFEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "vcpkg manifest of the engine")
  list(APPEND VCPKG_MANIFEST_FEATURES "gui")
  project(engine LANGUAGES CXX)

  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  add_subdirectory(../ECS ECS)
  add_subdirectory(../Editor Editor)
endif()

# Components: one header per component in Components/ (the struct and its
# E_REFLECT). A generated file includes them all, so every one of them is
# registered before main and offered by the editor's Add Component, without
# a list to maintain: adding a component is adding its header (CONFIGURE_DEPENDS
# picks it up at the next build).
file(GLOB_RECURSE ENGINE_COMPONENT_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Components/*.hpp)
set(ENGINE_COMPONENTS_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/Components.cpp")
set(_engine_components "// Generated by src/Engine/CMakeLists.txt: registers every component of src/Engine/Components.\n")
foreach(header IN LISTS ENGINE_COMPONENT_HEADERS)
  string(APPEND _engine_components "#include \"${header}\"\n")
endforeach()
# Written only when it changes, so it is not recompiled at every configure.
file(CONFIGURE OUTPUT "${ENGINE_COMPONENTS_SOURCE}" CONTENT "${_engine_components}" @ONLY)

add_executable(engine_gui main.cpp ${ENGINE_COMPONENTS_SOURCE})

# Includes from this folder: #include "Components/Position.hpp"
target_include_directories(engine_gui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})

# engine_editor brings engine_ecs, SDL3 and Dear ImGui with it; engine_ecs is
# listed too since main.cpp uses the ECS directly.
target_link_libraries(engine_gui PRIVATE
    engine_ecs
    engine_editor
)

# At the root of the build directory, next to the editor's executables: ./build/engine_gui
set_target_properties(engine_gui PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
